def batch_refine_communities(communities):
    import numpy as np
    out = {}
    if not isinstance(communities, list):
        return out
    for ci, c in enumerate(communities):
        nodes = c.get('nodes') if isinstance(c, dict) else None
        if not nodes:
            out[ci] = []
            continue
        S = set(int(x) for x in nodes if isinstance(x, (int, np.integer)) or (isinstance(x, bool) is False and str(x).lstrip("-").isdigit()))
        if not S:
            out[ci] = []
            continue
        adj = c.get('adjacency_list') if isinstance(c, dict) else None
        deg_inp = c.get('degree') if isinstance(c, dict) else None
        labels = c.get('node_labels') if isinstance(c, dict) else None
        if not isinstance(labels, dict):
            labels = None

        # Build internal neighbor lists, internal degrees
        nbrs_in = {}
        ideg = {}
        if isinstance(adj, dict):
            for u in S:
                nu = adj.get(u, [])
                if not isinstance(nu, (list, tuple, set, np.ndarray)):
                    nu = []
                lst = [int(v) for v in nu if v in S]
                nbrs_in[u] = lst
                ideg[u] = len(lst)
        else:
            nbrs_in = {u: [] for u in S}
            if isinstance(deg_inp, dict):
                ideg = {u: int(deg_inp.get(u, 0) or 0) for u in S}
            elif isinstance(deg_inp, (list, tuple, np.ndarray)) and len(deg_inp) > 0:
                nmax = max(S)
                arr = np.asarray(deg_inp)
                ideg = {u: int(arr[u]) if 0 <= u < len(arr) else 0 for u in S}
            else:
                ideg = {u: 0 for u in S}

        n = len(S)
        m2 = sum(ideg.values())
        avg_deg = (m2 / n) if n else 0.0
        if avg_deg < 1.0:
            avg_deg = 1.0

        # Label majority (within community)
        maj_label = None
        if labels:
            cnt = {}
            for u in S:
                lab = labels.get(u, None)
                if lab is None:
                    continue
                cnt[lab] = cnt.get(lab, 0) + 1
            if cnt:
                maj_label = max(cnt.items(), key=lambda x: x[1])[0]

        # Score each node (higher => more likely drop)
        drop = []
        # thresholds scaled for small communities; community size ~9 typical
        min_keep_deg = 1 if n <= 6 else 2
        deg_q25 = None
        try:
            deg_q25 = float(np.quantile(np.fromiter(ideg.values(), dtype=float), 0.25))
        except Exception:
            deg_q25 = None

        # Precompute neighbor sets for triangle checks only when adj given
        if isinstance(adj, dict):
            nbset = {u: set(nbrs_in[u]) for u in S}
        else:
            nbset = None

        for u in S:
            d = ideg.get(u, 0)
            # structural: leaf / very low internal degree
            low_deg = (d <= 1) or (d < min_keep_deg) or (deg_q25 is not None and d <= deg_q25 and d <= 2)
            # topological: weak triadic support (few triangles relative to possible wedges)
            tri_weak = False
            if nbset is not None and d >= 2:
                nu = nbrs_in[u]
                # sample pairs via min of neighbors to keep O(d^2) bounded for dense cases
                if len(nu) > 25:
                    nu = nu[:25]
                t = 0
                for i in range(len(nu)):
                    vi = nu[i]
                    s_vi = nbset.get(vi, set())
                    for j in range(i + 1, len(nu)):
                        if nu[j] in s_vi:
                            t += 1
                possible = (len(nu) * (len(nu) - 1)) / 2.0
                tri_weak = (possible > 0 and (t / possible) < 0.15)
            # label: mismatch + low neighbor agreement
            label_bad = False
            if labels and maj_label is not None:
                lu = labels.get(u, None)
                if lu is not None and lu != maj_label:
                    agree = 0
                    tot = 0
                    if nbset is not None:
                        for v in nbrs_in[u]:
                            lv = labels.get(v, None)
                            if lv is None:
                                continue
                            tot += 1
                            if lv == lu:
                                agree += 1
                    label_bad = (tot >= 2 and (agree / tot) < 0.34) or (d <= 1)

            # Combine signals: drop if multiple weak signals or extreme leaf/no-support
            bad = (low_deg and (tri_weak or label_bad)) or (d == 0) or (low_deg and n >= 8 and not labels)
            if bad:
                drop.append(u)

        out[ci] = sorted(drop)
    return out
